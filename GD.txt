import numpy as np
import matplotlib.pyplot as plt

def f(x):
    return (x + 3) ** 2

def derivative(x):
    return 2 * (x + 3)

x = 2
learning_rate = 0.1
iterations = 20

for i in range(iterations):
    x = x - learning_rate * derivative(x)
    print("Iteration:", i + 1, "x =", x, "y =", f(x))

print("Minimum value of x:", x)
print("Minimum value of y:", f(x))

x_values = np.linspace(-10, 5, 100)
y_values = f(x_values)

plt.plot(x_values, y_values)
plt.xlabel("x")
plt.ylabel("y")
plt.title("Gradient Descent: y = (x + 3)^2")
plt.show()
